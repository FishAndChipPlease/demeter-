#include #include #include #include // --- WiFi --- const char* WIFI_SSID = "YOUR_WIFI_NAME"; const char* WIFI_PASS = "YOUR_WIFI_PASSWORD"; // --- MQTT --- const char* BROKER = "broker.hivemq.com"; const int PORT = 1883; const char* TOPIC_SOIL = "demeter/soil"; const char* TOPIC_LIGHT = "demeter/light"; const char* TOPIC_RAIN = "demeter/rain"; const char* TOPIC_PUMP = "demeter/pump/status"; const char* TOPIC_PUMP_SET = "demeter/pump/set"; // --- Pins --- #define SOIL_PIN A0 #define LDR_PIN A1 #define RELAY_PIN 7 #define RAIN_PIN 8 // --- Soil calibration (adjust after testing!) --- const int SOIL_DRY = 620; // raw value when DRY const int SOIL_WET = 300; // raw value when WET // --- Objects --- LiquidCrystal_I2C lcd(0x27, 16, 2); // try 0x3F if blank WiFiClient wifi; MqttClient mqtt(wifi); // --- State --- bool pumpOn = false; bool isRaining = false; unsigned long lastSend = 0; unsigned long lastLCD = 0; int lcdPage = 0; // Rain debounce int rainStableCount = 0; bool rainRaw = false; void setup() { Serial.begin(9600); delay(1000); pinMode(RELAY_PIN, OUTPUT); digitalWrite(RELAY_PIN, HIGH); // pump OFF pinMode(RAIN_PIN, INPUT_PULLUP); lcd.init(); lcd.backlight(); lcd.setCursor(0, 0); lcd.print(" Demeter v1 "); lcd.setCursor(0, 1); lcd.print(" Starting up... "); delay(2000); // WiFi lcd.clear(); lcd.setCursor(0, 0); lcd.print("Connecting WiFi"); Serial.print("Connecting to WiFi..."); WiFi.begin(WIFI_SSID, WIFI_PASS); int tries = 0; while (WiFi.status() != WL_CONNECTED && tries < 30) { delay(500); Serial.print("."); tries++; } lcd.setCursor(0, 1); if (WiFi.status() == WL_CONNECTED) { Serial.println(" Connected!"); lcd.print("WiFi OK! "); } else { Serial.println(" Failed!"); lcd.print("WiFi FAILED! "); } delay(1500); // MQTT lcd.clear(); lcd.setCursor(0, 0); lcd.print("Connecting MQTT"); Serial.print("Connecting to MQTT..."); tries = 0; while (!mqtt.connect(BROKER, PORT) && tries < 10) { delay(1000); Serial.print("."); tries++; } lcd.setCursor(0, 1); if (mqtt.connected()) { Serial.println(" Connected!"); lcd.print("MQTT OK! "); } else { Serial.println(" Failed!"); lcd.print("MQTT FAILED! "); } delay(1500); mqtt.subscribe(TOPIC_PUMP_SET); lcd.clear(); lcd.setCursor(0, 0); lcd.print("Demeter Ready! "); lcd.setCursor(0, 1); lcd.print("Monitoring... "); delay(1500); } void loop() { mqtt.poll(); // --- Rain debounce --- bool nowRaining = (digitalRead(RAIN_PIN) == LOW); if (nowRaining == rainRaw) { if (rainStableCount < 20) rainStableCount++; } else { rainStableCount = 0; rainRaw = nowRaining; } if (rainStableCount >= 10 && isRaining != rainRaw) { isRaining = rainRaw; mqtt.beginMessage(TOPIC_RAIN); mqtt.print(isRaining ? "1" : "0"); mqtt.endMessage(); Serial.println(isRaining ? "RAIN detected!" : "Rain stopped"); if (isRaining && pumpOn) { pumpOn = false; digitalWrite(RELAY_PIN, HIGH); mqtt.beginMessage(TOPIC_PUMP); mqtt.print("0"); mqtt.endMessage(); } } // --- Pump command from app --- if (mqtt.available()) { String topic = mqtt.messageTopic(); String msg = ""; while (mqtt.available()) msg += (char)mqtt.read(); if (topic == TOPIC_PUMP_SET) { pumpOn = (msg == "1"); digitalWrite(RELAY_PIN, pumpOn ? LOW : HIGH); mqtt.beginMessage(TOPIC_PUMP); mqtt.print(pumpOn ? "1" : "0"); mqtt.endMessage(); Serial.println(pumpOn ? "Pump ON" : "Pump OFF"); lcd.clear(); lcd.setCursor(0, 0); lcd.print(" Water Pump "); lcd.setCursor(0, 1); lcd.print(pumpOn ? " >> ON << " : " >> OFF << "); } } // --- Send sensors every 3s --- if (millis() - lastSend > 3000) { lastSend = millis(); int rawSoil = analogRead(SOIL_PIN); int soil = constrain(map(rawSoil, SOIL_DRY, SOIL_WET, 0, 100), 0, 100); int light = map(analogRead(LDR_PIN), 0, 1023, 0, 100); if (soil < 30 && !pumpOn && !isRaining) { pumpOn = true; digitalWrite(RELAY_PIN, LOW); mqtt.beginMessage(TOPIC_PUMP); mqtt.print("1"); mqtt.endMessage(); Serial.println("Auto watering!"); } mqtt.beginMessage(TOPIC_SOIL); mqtt.print(soil); mqtt.endMessage(); mqtt.beginMessage(TOPIC_LIGHT); mqtt.print(light); mqtt.endMessage(); Serial.print("Soil:"); Serial.print(soil); Serial.print("% (raw "); Serial.print(rawSoil); Serial.print(")"); Serial.print(" Light:"); Serial.print(light); Serial.print("% Rain:"); Serial.println(isRaining ? "YES" : "no"); } // --- Rotate LCD every 3s --- if (millis() - lastLCD > 3000 && !pumpOn) { lastLCD = millis(); int rawSoil = analogRead(SOIL_PIN); int soil = constrain(map(rawSoil, SOIL_DRY, SOIL_WET, 0, 100), 0, 100); int light = map(analogRead(LDR_PIN), 0, 1023, 0, 100); lcd.clear(); if (isRaining) { lcd.setCursor(0, 0); lcd.print("** RAINING ** "); lcd.setCursor(0, 1); lcd.print("Watering paused "); } else { switch (lcdPage) { case 0: lcd.setCursor(0, 0); lcd.print("Soil Moisture:"); lcd.setCursor(0, 1); lcd.print(soil); lcd.print("% "); lcd.print(soil < 30 ? "DRY!" : soil < 60 ? "Moist" : "Wet"); break; case 1: lcd.setCursor(0, 0); lcd.print("Light Level:"); lcd.setCursor(0, 1); lcd.print(light); lcd.print("% "); lcd.print(light < 30 ? "Dim" : light < 70 ? "Moderate" : "Bright"); break; case 2: lcd.setCursor(0, 0); lcd.print("Pump: "); lcd.print(pumpOn ? "ON " : "OFF"); lcd.setCursor(0, 1); lcd.print("Auto mode: ON"); break; } lcdPage = (lcdPage + 1) % 3; } } }
